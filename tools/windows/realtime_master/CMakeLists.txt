cmake_minimum_required(VERSION 3.20)
project(hako_realtime_master LANGUAGES CXX)
# runtime の tools/master_runner と同じ構成（そちらは変更しない）。違いは壁時計に合わせて進めること（main.cpp）
find_package(hakoniwa-core CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(hako-realtime-master main.cpp)
target_compile_features(hako-realtime-master PRIVATE cxx_std_20)
target_link_libraries(hako-realtime-master PRIVATE hakoniwa-core::hako Threads::Threads)
get_target_property(HAKO_CORE_INCLUDE_DIRS hakoniwa-core::conductor INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(hako-realtime-master PRIVATE ${HAKO_CORE_INCLUDE_DIRS})
if(WIN32)
    target_compile_definitions(hako-realtime-master PRIVATE WIN32 NOMINMAX)
    target_link_libraries(hako-realtime-master PRIVATE winmm)
endif()
if(MSVC)
    target_compile_options(hako-realtime-master PRIVATE /utf-8)
endif()
install(TARGETS hako-realtime-master RUNTIME DESTINATION bin)
