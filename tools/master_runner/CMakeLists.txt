cmake_minimum_required(VERSION 3.20)
project(hako_runtime_master_runner LANGUAGES CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR OR CMAKE_BINARY_DIR MATCHES "[/\\\\]third_party[/\\\\]")
    message(FATAL_ERROR "Use an out-of-source build outside third_party")
endif()
find_package(hakoniwa-core CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(hako-runtime-master main.cpp)
target_compile_features(hako-runtime-master PRIVATE cxx_std_20)
target_link_libraries(hako-runtime-master PRIVATE hakoniwa-core::hako Threads::Threads)
# The upstream export gives hakoniwa-core::hako no include directory; reuse the package's.
get_target_property(HAKO_CORE_INCLUDE_DIRS hakoniwa-core::conductor INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(hako-runtime-master PRIVATE ${HAKO_CORE_INCLUDE_DIRS})
if(WIN32)
    target_compile_definitions(hako-runtime-master PRIVATE WIN32 NOMINMAX)
endif()
if(MSVC)
    target_compile_options(hako-runtime-master PRIVATE /utf-8)
endif()
install(TARGETS hako-runtime-master RUNTIME DESTINATION bin)
